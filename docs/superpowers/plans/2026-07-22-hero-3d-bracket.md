# Hero 3D Bracket Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the flat, low-opacity `</>` background glyph in the Hero section with a real, opaque 3D `</>` object (built from `three`/`@react-three/fiber` primitives) that tilts toward the cursor across the whole hero section, sitting in its own column with a clear gap from the text, and hidden entirely below the `lg` breakpoint.

**Architecture:** A new presentational component (`Hero3DBracket`) owns the WebGL canvas, geometry, and per-frame tilt animation. `Hero.tsx` switches from a single text column to a two-column layout, tracks pointer position across the whole section via a ref (no re-renders), and only mounts `Hero3DBracket` on desktop-width viewports.

**Tech Stack:** React 19.2, TypeScript (strict), Vite, Tailwind v4, `three`, `@react-three/fiber`.

## Global Constraints

- This repo has no test runner or linter — `npm run build` (`tsc -b`, strict mode) is the only automated check. Every task's verification step is a successful `npm run build`, plus a manual dev-server check for the task that changes visible behavior.
- Components never hardcode copy — none of this work touches copy, so `content.ts` is not modified. The existing `options.showCodeMotif` toggle (already in `content.ts`) is reused as-is to gate the new 3D object; no new option is added.
- Installed React is `19.2.8`. `@react-three/fiber@9.6.1`'s peer range is `react: ">=19 <19.3"` — compatible, no `--legacy-peer-deps` needed.
- No `@react-three/drei` — the bracket is hand-built from `three` primitives, not `drei` helpers or extruded font text.
- Follow existing style conventions: Tailwind utility classes mixed with inline `style` objects for `clamp()`-based fluid sizing (see `Hero.tsx` today).

- [ ] **Step 1: Write the component**

Create `src/components/Hero3DBracket.tsx`:

```tsx
import { useRef, type RefObject } from "react";
import { Canvas, useFrame } from "@react-three/fiber";
import * as THREE from "three";

const PRIMARY = "#f2f207";
const SECONDARY = "#078cf2";
const CHEVRON_THETA = Math.PI / 5.5;
const ARM_LENGTH = 1.3;
const ARM_THICKNESS = 0.24;

export type Pointer = { x: number; y: number };

function ChevronArm({ angle, color }: { angle: number; color: string }) {
  return (
    <mesh
      position={[
        -(ARM_LENGTH / 2) * Math.sin(angle),
        (ARM_LENGTH / 2) * Math.cos(angle),
        0,
      ]}
      rotation={[0, 0, angle]}
    >
      <boxGeometry args={[ARM_THICKNESS, ARM_LENGTH, ARM_THICKNESS]} />
      <meshStandardMaterial color={color} roughness={0.35} metalness={0.1} />
    </mesh>
  );
}

function Chevron({
  position,
  mirrored,
}: {
  position: [number, number, number];
  mirrored: boolean;
}) {
  return (
    <group position={position} scale={[mirrored ? -1 : 1, 1, 1]}>
      <ChevronArm angle={-CHEVRON_THETA} color={PRIMARY} />
      <group scale={[1, -1, 1]}>
        <ChevronArm angle={-CHEVRON_THETA} color={PRIMARY} />
      </group>
    </group>
  );
}

function Slash() {
  return (
    <mesh rotation={[0, 0, -Math.PI / 7]}>
      <boxGeometry args={[ARM_THICKNESS, 2.4, ARM_THICKNESS]} />
      <meshStandardMaterial color={SECONDARY} roughness={0.35} metalness={0.1} />
    </mesh>
  );
}

function Bracket({ pointerRef }: { pointerRef: RefObject<Pointer> }) {
  const group = useRef<THREE.Group>(null);
  const reducedMotion =
    typeof window !== "undefined" &&
    window.matchMedia("(prefers-reduced-motion: reduce)").matches;

  useFrame(() => {
    if (reducedMotion || !group.current) return;
    const targetX = (pointerRef.current.y * Math.PI) / 10;
    const targetY = (pointerRef.current.x * Math.PI) / 8;
    group.current.rotation.x = THREE.MathUtils.lerp(
      group.current.rotation.x,
      targetX,
      0.06,
    );
    group.current.rotation.y = THREE.MathUtils.lerp(
      group.current.rotation.y,
      targetY,
      0.06,
    );
  });

  return (
    <group ref={group}>
      <Chevron position={[-1.6, 0, 0]} mirrored={false} />
      <Slash />
      <Chevron position={[1.6, 0, 0]} mirrored={true} />
    </group>
  );
}

export default function Hero3DBracket({
  pointerRef,
}: {
  pointerRef: RefObject<Pointer>;
}) {
  return (
    <Canvas
      gl={{ alpha: true, antialias: true }}
      camera={{ position: [0, 0, 6], fov: 40 }}
      style={{ width: "100%", height: "100%" }}
    >
      <ambientLight intensity={0.65} />
      <directionalLight position={[3, 4, 5]} intensity={1.1} />
      <Bracket pointerRef={pointerRef} />
    </Canvas>
  );
}
```

import { useEffect, useRef, useState } from "react";
import { content, options } from "../content";
import Hero3DBracket, { type Pointer } from "./Hero3DBracket";

function useIsDesktop(minWidth = 1024) {
  const [isDesktop, setIsDesktop] = useState(
    () => typeof window !== "undefined" && window.matchMedia(`(min-width: ${minWidth}px)`).matches,
  );

  useEffect(() => {
    const query = window.matchMedia(`(min-width: ${minWidth}px)`);
    setIsDesktop(query.matches);
    const handleChange = (event: MediaQueryListEvent) => setIsDesktop(event.matches);
    query.addEventListener("change", handleChange);
    return () => query.removeEventListener("change", handleChange);
  }, [minWidth]);

  return isDesktop;
}

export default function Hero() {
  const isDesktop = useIsDesktop();
  const pointerRef = useRef<Pointer>({ x: 0, y: 0 });

  function handlePointerMove(event: React.PointerEvent<HTMLElement>) {
    const rect = event.currentTarget.getBoundingClientRect();
    pointerRef.current.x = ((event.clientX - rect.left) / rect.width) * 2 - 1;
    pointerRef.current.y = ((event.clientY - rect.top) / rect.height) * 2 - 1;
  }

  return (
    <section
      onPointerMove={handlePointerMove}
      className="relative mx-auto flex w-full flex-1 flex-col justify-center lg:grid lg:items-center"
      style={{
        padding: "clamp(28px,5vw,56px) clamp(20px,5vw,64px)",
        maxWidth: 1280,
        gridTemplateColumns: "minmax(0,1fr) clamp(260px,28vw,420px)",
        columnGap: "clamp(32px,5vw,64px)",
      }}
    >
      {/* code motif: dot-grid (decorative) */}
      {options.showCodeMotif && (
        <div
          aria-hidden="true"
          className="dot-grid pointer-events-none absolute inset-0 overflow-hidden"
        />
      )}

      <div
        className="relative flex flex-col"
        style={{ gap: "clamp(20px,3vw,34px)", maxWidth: 900 }}
      >
        <div
          className="font-mono uppercase text-secondary"
          style={{
            fontSize: "clamp(12px,1.4vw,14px)",
            letterSpacing: "0.22em",
          }}
        >
          {content.eyebrow}
        </div>

        <h1
          className="font-display"
          style={{
            fontWeight: 800,
            fontSize: "clamp(52px,11vw,132px)",
            lineHeight: 0.92,
            letterSpacing: "-0.03em",
            textWrap: "balance",
          }}
        >
          {content.name}
        </h1>

        <p
          style={{
            fontSize: "clamp(18px,2.4vw,28px)",
            lineHeight: 1.4,
            maxWidth: 660,
            color: "oklch(0.82 0.008 250)",
            textWrap: "pretty",
          }}
        >
          {content.intro.lead}
          <span className="text-primary" style={{ fontWeight: 600 }}>
            {content.intro.highlight}
          </span>
          {content.intro.tail}
        </p>

        <div className="flex flex-wrap" style={{ gap: 14, marginTop: 8 }}>
          <a
            href="#work"
            className="rounded-full bg-primary font-mono text-black transition-transform hover:-translate-y-0.5 hover:text-black"
            style={{ fontSize: 15, fontWeight: 500, padding: "15px 28px" }}

            Check out my work →
          </a>
          <a

            style={{ fontSize: 15, padding: "15px 28px" }}

            Let's chat
          </a>
        </div>
      </div>

      {options.showCodeMotif && isDesktop && (
        <div
          className="relative"
          style={{ height: "clamp(260px,28vw,420px)" }}
        >
          <Hero3DBracket pointerRef={pointerRef} />
        </div>
      )}
    </section>
  );
}
